cmake_minimum_required(VERSION 3.24)

project(muxmender_d3d11 LANGUAGES CXX)

add_executable(muxmender-d3d11 src/main.cpp)
target_compile_features(muxmender-d3d11 PRIVATE cxx_std_20)
target_compile_definitions(muxmender-d3d11 PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(muxmender-d3d11 PRIVATE d3d11 dxgi)

if(MSVC)
  target_compile_options(muxmender-d3d11 PRIVATE /W4 /permissive- /EHsc)
endif()

option(MUXMENDER_COLOR_TEST "Build the generated-color D3D11 smoke test" OFF)
option(MUXMENDER_DV_PREVIEW "Build experimental DV profile 5 SDR preview" OFF)
if(MUXMENDER_DV_PREVIEW)
  enable_language(C)
  set(PLACEBO_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/build/_deps/libplacebo-src" CACHE PATH "libplacebo source")
  set(PLACEBO_BUILD "${CMAKE_CURRENT_SOURCE_DIR}/build/_deps/libplacebo-build-clang" CACHE PATH "libplacebo build")
  set(FFMPEG_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/vcpkg_installed/x64-windows-release" CACHE PATH "FFmpeg installation")
  add_executable(muxmender-dv-preview src/dv_preview.cpp src/libav_bridge.c)
  target_compile_features(muxmender-dv-preview PRIVATE cxx_std_20 c_std_11)
  target_compile_definitions(muxmender-dv-preview PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_include_directories(muxmender-dv-preview PRIVATE "${PLACEBO_SOURCE}/src/include" "${PLACEBO_BUILD}/src/include" "${FFMPEG_ROOT}/include")
  target_link_libraries(muxmender-dv-preview PRIVATE "${PLACEBO_BUILD}/src/libplacebo.lib")
  foreach(lib avformat avcodec avutil swscale)
    target_link_libraries(muxmender-dv-preview PRIVATE "${FFMPEG_ROOT}/lib/${lib}.lib")
  endforeach()
  file(GLOB FFMPEG_DLLS "${FFMPEG_ROOT}/bin/av*.dll" "${FFMPEG_ROOT}/bin/sw*.dll")
  add_custom_command(TARGET muxmender-dv-preview POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different ${FFMPEG_DLLS} "${PLACEBO_BUILD}/src/libplacebo-371.dll" "$<TARGET_FILE_DIR:muxmender-dv-preview>")
endif()
if(MUXMENDER_COLOR_TEST)
  set(PLACEBO_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/build/_deps/libplacebo-src" CACHE PATH "libplacebo source")
  set(PLACEBO_BUILD "${CMAKE_CURRENT_SOURCE_DIR}/build/_deps/libplacebo-build-clang" CACHE PATH "libplacebo build")
  add_executable(muxmender-color-test src/color_test.cpp)
  target_compile_features(muxmender-color-test PRIVATE cxx_std_20)
  target_compile_definitions(muxmender-color-test PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_include_directories(muxmender-color-test PRIVATE "${PLACEBO_SOURCE}/src/include" "${PLACEBO_BUILD}/src/include")
  target_link_libraries(muxmender-color-test PRIVATE "${PLACEBO_BUILD}/src/libplacebo.lib")
  add_custom_command(TARGET muxmender-color-test POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${PLACEBO_BUILD}/src/libplacebo-371.dll" "$<TARGET_FILE_DIR:muxmender-color-test>")
  enable_testing()
  add_test(NAME generated-colors COMMAND muxmender-color-test)
  set_tests_properties(generated-colors PROPERTIES TIMEOUT 30)
endif()
